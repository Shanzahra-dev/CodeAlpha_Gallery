async function getCommonsCategory(category,title,target=1100){
  const found=[];const seen=new Set();let cont={};let attempts=0;
  while(found.length<target && attempts++<20){
    const params=new URLSearchParams({action:'query',format:'json',origin:'*',generator:'categorymembers',gcmtitle:title,gcmtype:'file',gcmlimit:'500',prop:'imageinfo',iiprop:'url|mime',iiurlwidth:'1000',...cont});
    const response=await fetch('https://commons.wikimedia.org/w/api.php?'+params.toString());
    if(!response.ok)throw Error('Wikimedia Commons request failed');
    const data=await response.json();
    for(const page of Object.values(data.query?.pages||{})){
      const info=page.imageinfo?.[0];const mime=info?.mime||'';
      if(!['image/jpeg','image/png','image/webp'].includes(mime)||!info?.thumburl||seen.has(page.pageid))continue;
      seen.add(page.pageid);
      const label=page.title.replace(/^File:/,'').replace(/\.[^.]+$/,'').replace(/[_-]/g,' ').replace(/\s+/g,' ').trim();
      found.push({id:'commons-'+page.pageid,title:label.slice(0,75),category,type:mime==='image/jpeg'?'jpg':mime==='image/png'?'png':'webp',tags:label.toLowerCase(),ratio:'4/5',url:info.thumburl});
      if(found.length>=target)break;
    }
    if(!data.continue)break;
    cont=data.continue;
  }
  return found;
}
async function loadThousandPhotos(){
  let loaded=0;
  const summary={};
  for(const [category,titles] of Object.entries(commonsCategories)){
    let categoryLoaded=0;
    const existing=new Set(photos.filter(p=>p.category===category).map(p=>p.id));
    for(const title of titles){
      if(categoryLoaded>=TARGET_PER_CATEGORY)break;
      try{
        statusLine.textContent=`Loading ${category}: ${categoryLoaded} / ${TARGET_PER_CATEGORY} additional images · ${loaded} total fetched`;
        const batch=await getCommonsCategory(category,title,TARGET_PER_CATEGORY-categoryLoaded);
        for(const photo of batch){
          if(!existing.has(photo.id)){
            photos.push(photo);existing.add(photo.id);categoryLoaded++;loaded++;
          }
        }
        renderGallery();
        // Yield between collections to keep the page responsive.
        await new Promise(resolve=>setTimeout(resolve,0));
      }catch(error){console.warn('Could not load Commons collection',title,error);}
    }
    summary[category]=categoryLoaded;
    statusLine.textContent=`${category} loaded ${categoryLoaded} new images. Loading remaining categories…`;
  }
  statusLine.textContent=`Fetched ${loaded} additional photos: `+Object.entries(summary).map(([k,v])=>`${k} ${v}`).join(' · ')+'. Actual counts depend on Wikimedia Commons availability.';
  renderGallery();
}
